<%*
const types = ["Лекция", "Практика", "ДЗ"];
const selType = await tp.system.suggester(types, types, false, "Тип занятия?");

const subjects = ["Дискретка", "Матан", "Алгем", "ДопАлгем", "Прога", "Основы ИИ", "Фронтиры ИИ", "Англ", "Русский", "Письмо", "История", "ОРГ"];
const folders = ["Математика/Дискретка", "Математика/Матан", "Математика/Алгем", "Математика/ДопАлгем", "IT/Прога", "IT/Основы ИИ", "IT/Фронтиры ИИ", "Общеобразовательные/Англ", "Общеобразовательные/Русский", "Общеобразовательные/Письмо", "Общеобразовательные/История", "Общеобразовательные/ОРГ"];
const selFolder = await tp.system.suggester(subjects, folders, false, "Предмет?");

const topic = await tp.system.prompt("Тема (можно оставить пустым):");
const titleSuffix = topic ? ` - ${topic}` : "";

const dateStr = tp.date.now("DD.MM.YYYY");
const newName = `${dateStr} - ${selType}${titleSuffix}`;

await tp.file.move(`${selFolder}/${newName}`);

const status = (selType === "ДЗ") ? "долг" : "в_процессе";
-%>
---
type: <% selType.toLowerCase() %>
status: <% status %>
date: <% dateStr %>
tags:
  - 
---
# <% selType %> - <% dateStr %><% titleSuffix %>
